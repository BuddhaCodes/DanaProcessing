Runnable example

A complete sketch using this section. Test in Dana opens it in the IDE, or paste it into a new tab and press Run.

// Map, Constrain, Lerp, Dist and trig: a ball that eases toward the mouse.
public class Easing : Sketch
{
    float x = 300, y = 200;

    public override void Setup() => Size(600, 400);

    public override void Draw()
    {
        Background(240);
        x = Lerp(x, MouseX, 0.08f);
        y = Lerp(y, MouseY, 0.08f);

        float d = Dist(x, y, MouseX, MouseY);
        float size = Constrain(Map(d, 0, 200, 20, 80), 20, 80);
        Fill(60, 120, 200); NoStroke();
        Circle(x, y, size);

        // Atan2 gives the angle toward the mouse.
        float a = Atan2(MouseY - y, MouseX - x);
        Stroke(200, 60, 60); StrokeWeight(3);
        Line(x, y, x + Cos(a) * 60, y + Sin(a) * 60);

        Fill(30);
        Text($"distance {d:0} · angle {Degrees(a):0}°", 10, 20);
    }
}
